===== E:\learn web development\hotel reception\src\middleware\globaValidator.js =====

export const validator = (schema)=>{
    return (req , res , next)=>{
        let errors = []
        for( let key of ['body' , 'params' , 'query'] ){
            if(!req[key] || !schema[key] ){continue;}
            const {error , value} = schema[key].validate(req[key],{
                abortEarly: false,
                stripUnknown: true,
                allowUnknown: false
            })
            if(error?.details?.length>0){
                error.details.forEach(element => {
                    errors.push({
                        field:element.path.join('.'),
                        message:element.message})
                });
            }else{
                req[key]=value;
            }
        }

        if(errors.length>0){
            return res.status(400).json({success: false , message: "validator error", errors})
        }

        next();
    }
}

===== E:\learn web development\hotel reception\src\middleware\verifyJWT.js =====
import jwt from 'jsonwebtoken'
import { AppError } from '../util/AppError.js';
import { User } from '../models/user.js';
import { env } from '../config/env.js';
export const verifyJWT = async(req,res,next)=>{
    const header = req.headers.authorization;
    if(!header||!header.startsWith('Bearer')){
        throw new AppError('unauthorized',401);
    }
    const token = header.split(' ')[1];
    let decoded
    try{
        decoded = jwt.verify(token,env.ACCESS_TOKEN_SECRET);
    }catch(err){
        throw new AppError('invalid or expired token', 401);
    }
    const id = decoded.sub;
    const foundUser =await User.findById(id);
    if(!foundUser){
        throw new AppError('user not found',401)
    }
    if(foundUser.isActive=== false){
        throw new AppError('user is not active',401)
    }
    req.user= foundUser
    next();
}

import { User } from "../../models/user.js";
import { AppError } from "../../util/AppError.js";
import {env} from '../../config/env.js'
import jwt from 'jsonwebtoken'
import bcrypt from 'bcrypt'
export const login =async (req,res)=>{
    const {email , password} = req.body;
    const foundUser =await User.findOne({email}).select('+passwordHash')
    if(!foundUser || foundUser?.isActive === false){
        throw new AppError("Invalid credentials", 401);
    }
    const confirm =await bcrypt.compare(password , foundUser.passwordHash);
    if(!confirm){
        throw new AppError("Invalid credentials", 401);
    }
    const accessToken = jwt.sign({sub:foundUser._id,role:foundUser.role}, env.ACCESS_TOKEN_SECRET,{
        expiresIn: env.ACCESS_TOKEN_EXPIRES_IN
    })
    const refreshToken = jwt.sign({sub:foundUser._id,role:foundUser.role},env.REFRESH_TOKEN_SECRET,{
        expiresIn: env.REFRESH_TOKEN_EXPIRES_IN
    })
    res.cookie("refreshToken",refreshToken,{
        httpOnly: true,
        secure: env.NODE_ENV === "production"
    });
    res.status(200).json({
        success:true,
        message:"Logged in successfully",
        data:{
            accessToken,
            tokenType:"Bearer",
            expiresIn: env.ACCESS_TOKEN_EXPIRES_IN,
            user: {id: foundUser._id, name : foundUser.name, email : foundUser.email ,role : foundUser.role}
        }
    })
}
